Scatter Plot with Comments

# Note the incremental additions
x <- c(0.5, 2, 4, 8, 12, 16)
y1 <- c(1, 1.3, 1.9, 3.4, 3.9, 4.8)
y2 <- c(4, .8, .5, .45, .4, .3)

# Setting label orientation, margins c(bottom, left, top, right) & text size
par(las=1, mar=c(4, 4, 2, 4), cex=.7)
## las=1: makes axis tick labels always sit horizontal, easier to read than rotated numbers
## mar: sets the four margins so there is room for axis titles on all four sides, including the right
## cex=.7: shrinks all text in the plot to 70% size, so labels don't crowd the small panel


# Opens a blank plotting region, with no data or axes drawn yet, this is the "new canvas" step
plot.new() 


# Defines the coordinate system (x from min to max of x, y from 0 to 6), without drawing anything visible,
plot.window(range(x), c(0, 6))  
## this fixes the scale so every later call plots into the same space


# Draws a connecting line through the y1 points, no markers yet, just the path
lines(x, y1) 

# Draws a second connecting line through the y2 points, so both trends are visible before markers go on
lines(x, y2)  

  
# Adds solid filled circles at each y1 coordinate, 
points(x, y1, pch=16, cex=2)
## pch=16 is a solid dot, cex=2 makes them large enough to stand out over the line


# Adds hollow circles (pch=21 supports a fill) at each y2 coordinate
points(x, y2, pch=21, bg="white", cex=2)
## bg="white" keeps the center empty so this series looks visually distinct from y1's solid dots


# Switches the default drawing color, the plot foreground (fg, used for axis lines and box), and axis label color all to gray
par(col="gray50", fg="gray50", col.axis="gray50")  
## axes/box that come next look secondary to the data itself

# Draws the bottom axis (1 = bottom side) with tick marks at 0, 4, 8, etc.
axis(1, at=seq(0, 16, 4)) 
## the "1" specifies which side of the plot the axis is drawn on


# Draws the left axis (2 = left side) with ticks at 0,2,4,6, corresponding to the y1 series' scale
axis(2, at=seq(0, 6, 2))  


# Draws a second axis on the right side (4 = right side) with the same tick positions
axis(4, at=seq(0, 6, 2))  
## This lets the chart show a second labeled scale, used later for the y2 series

 
# Draws a box around the plotting region, bty="u" means a "U shaped" box, open at the top, since this chart doesn't need a full rectangle
box(bty="u") 


# Adds the x-axis title below the bottom axis (side=1), line=2 controls how far from the axis it sits, cex=0.8 sets its size
mtext("Travel Time (s)", side=1, line=2, cex=0.8) 


# Adds the left y-axis title, las=0 lets this text rotate to run vertically alongside the axis, matching the traditional R default orientation for axis titles
mtext("Responses per Travel", side=2, line=2, las=0, cex=0.8)  


# Adds a title for the right-side axis, giving that second scale its own label, tying back to the y2 series
mtext("Responses per Second", side=4, line=2, las=0, cex=0.8)  


# Places a text annotation directly on the plot at coordinates (4, 5), used here as an informal title/label naming the data source, rather than a proper axis title
text(4, 5, "Bird 131")  
# Resets the margins and all the colors back to R's defaults
par(mar=c(5.1, 4.1, 4.1, 2.1), col="black", fg="black", col.axis="black")
## The later plots in the same script aren't affected by the gray/narrow-margin settings made for this one

# Reshape to long format since ggplot2 wants one row per observation, not wide columns
df <- data.frame(
  x = rep(x, 2),
  y = c(y1, y2),
  series = rep(c("Responses per Travel", "Responses per Second"), each = length(x))
)


ggplot(df, aes(x = x, y = y, color = series, shape = series)) +
  geom_line(linewidth = 0.5) +
  geom_point(size = 3) +
  scale_color_manual(values = c("Responses per Travel" = "darkgreen",
                                 "Responses per Second" = "darkorange")) +
  scale_shape_manual(values = c("Responses per Travel" = 16,
                                 "Responses per Second" = 17)) +
  scale_x_continuous(breaks = seq(0, 16, 4)) +
  scale_y_continuous(breaks = seq(0, 6, 2),
                      sec.axis = dup_axis(name = "Responses per Second")) +
  labs(x = "Travel Time (s)", y = "Responses per Travel") +
  annotate("text", x = 4, y = 5, label = "Bird 131",
           family = "serif", fontface = "italic", size = 3) +
  theme_minimal(base_family = "serif", base_size = 10) +
  theme(
    axis.text = element_text(color = "gray40"),
    axis.ticks = element_line(color = "gray40"),
    legend.position = "none"
  )


Mapping: Base R to ggplot2

  • family="serif" → theme(base_family = "serif")
  • col="darkgreen" / "darkorange" → scale_color_manual()
  • pch=16 / pch=17 → scale_shape_manual()
  • cex=.7 → roughly base_size= in theme_minimal()
  • col.axis="gray40" → theme(axis.text = element_text(color="gray40"))
  • axis(4, ...) + mtext(..., side=4) → scale_y_continuous(sec.axis = dup_axis(...))
  • mtext() axis titles → labs(x=, y=)
  • text(4, 5, "Bird 131", font=3) → annotate("text", ..., fontface="italic")
  • las=1 (horizontal tick labels) → no direct equivalent needed, ggplot2 draws horizontal labels by default
  • box(bty="u") (open-top box) → no equivalent, ggplot2’s panel.border is all-or-nothing; a partial “U” box would need manually drawn line segments
  • mar=c(4,4,2,4) → theme(plot.margin = margin(...)), though ggplot2 manages spacing automatically so this is rarely needed

# Removing Russia since it seems to be absent in the graph template
df <- dplyr::filter(owideu,
                    !is.na(new_deaths),
                    location != "Russia")

Calculating the quartiles and IQR for outliers

q3 <- quantile(df$new_deaths, 0.75, na.rm=T)
iqr <- IQR(df$new_deaths)
sup <- q3 + 1.5*iqr
df$outliers <- ifelse(df$new_deaths > sup, df$location, NA)
