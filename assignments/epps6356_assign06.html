                  max = 50,

    mainPanel(

      # Output: Histogram ----
      plotOutput(outputId = "distPlot")

    )
  )
)

# Define server logic required to draw a histogram ----
server <- function(input, output) {

  # Histogram of the Old Faithful Geyser Data ----
  # with requested number of bins
  # This expression that generates a histogram is wrapped in a call
  # to renderPlot to indicate that:
  #
  # 1. It is "reactive" and therefore should be automatically
  #    re-executed when inputs (input$bins) change
  # 2. Its output type is a plot
  output$distPlot <- renderPlot({

    x    <- faithful$waiting
    bins <- seq(min(x), max(x), length.out = input$bins + 1)

    hist(x, breaks = bins, col = "#75AADB", border = "white",
         xlab = "Waiting time to next eruption (in mins)",
         main = "Histogram of waiting times")

    })

}

# Create Shiny app ----
shinyApp(ui = ui, server = server)

# Define UI for slider demo app ----
ui <- fluidPage(

  # App title ----
  titlePanel("Sliders"),

  # Sidebar layout with input and output definitions ----
  sidebarLayout(

    # Sidebar to demonstrate various slider options ----
    sidebarPanel(

      # Input: Simple integer interval ----
      sliderInput("integer", "Integer:",
                  min = 0, max = 1000,
                  value = 500),

      # Input: Decimal interval with step value ----
      sliderInput("decimal", "Decimal:",
                  min = 0, max = 1,
                  value = 0.5, step = 0.1),

      # Input: Specification of range within an interval ----
      sliderInput("range", "Range:",
                  min = 1, max = 1000,
                  value = c(200,500)),

      # Input: Custom currency format for with basic animation ----
      sliderInput("format", "Custom Format:",
                  min = 0, max = 10000,
                  value = 0, step = 2500,
                  pre = "$", sep = ",",
                  animate = TRUE),

      # Input: Animation with custom interval (in ms) ----
      # to control speed, plus looping
      sliderInput("animation", "Looping Animation:",
                  min = 1, max = 2000,
                  value = 1, step = 10,
                  animate =
                    animationOptions(interval = 300, loop = TRUE))

    ),

    # Main panel for displaying outputs ----
    mainPanel(

      # Output: Table summarizing the values entered ----
      tableOutput("values")

    )
  )
)

# Define server logic for slider examples ----
server <- function(input, output) {

  # Reactive expression to create data frame of all input values ----
  sliderValues <- reactive({

    data.frame(
      Name = c("Integer",
               "Decimal",
               "Range",
               "Custom Format",
               "Animation"),
      Value = as.character(c(input$integer,
                             input$decimal,
                             paste(input$range, collapse = " "),
                             input$format,
                             input$animation)),
      stringsAsFactors = FALSE)

  })

  # Show the values in an HTML table ----
  output$values <- renderTable({
    sliderValues()
  })

}

# Create Shiny app ----
shinyApp(ui, server)

Review

which method would you use for a real project, and why?

  • I’d use the bslib + thematic method for a real project. It keeps font and color choices centralized in one bs_theme() call instead of scattered CSS strings, it’s reusable across multiple apps, and thematic_shiny() extends the theme to plots automatically — which the CSS-only approach can’t do without manually restyling every plot call.

mtcars

ui <- fluidPage(
  titlePanel("Motor Trend Cars: Fuel Efficiency Explorer"),

  sidebarLayout(
    sidebarPanel(
      selectInput("xvar", "X-axis variable:",
                  choices = c("Weight (1000 lbs)" = "wt",
                              "Horsepower" = "hp",
                              "Displacement (cu.in.)" = "disp"),
                  selected = "wt"),
      checkboxGroupInput("cyl", "Number of cylinders:",
                          choices = c(4, 6, 8),
                          selected = c(4, 6, 8))
    ),

    mainPanel(
      plotOutput("scatterPlot"),
      tableOutput("summaryTable")
    )
  )
)

server <- function(input, output) {

  filtered_data <- reactive({
    req(input$cyl)
    mtcars %>% filter(cyl %in% input$cyl)
  })

  output$scatterPlot <- renderPlot({
    df <- filtered_data()

    xlab_text <- switch(input$xvar,
                         wt   = "Weight (1000 lbs)",
                         hp   = "Horsepower",
                         disp = "Displacement (cu. in.)")

    ggplot(df, aes(x = .data[[input$xvar]], y = mpg, color = factor(cyl))) +
      geom_point(size = 3) +
      scale_color_manual(values = c("4" = "#E07B32", "6" = "#2E8B51", "8" = "#6A00AD")) +
      labs(title = "Fuel Efficiency vs. Selected Variable",
         x = xlab_text,
         y = "Miles per Gallon (mpg)",
         color = "Cylinders",
         caption = "Source: mtcars dataset, 1974 Motor Trend US magazine") +
      theme_minimal()
  })

  output$summaryTable <- renderTable({
    filtered_data() %>%
      group_by(cyl) %>%
      summarise(
        `Mean MPG` = round(mean(mpg), 1),
        `Mean HP` = round(mean(hp), 1),
        `Mean Weight (1000 lbs)` = round(mean(wt), 2),
        `N cars` = n()
      )
  })

}

us_arrests <- USArrests %>%
  rownames_to_column(var = "State")

ui <- fluidPage(
  titlePanel("US Arrests (1973): State Crime Rates Explorer"),

  sidebarLayout(
    sidebarPanel(
      selectInput("stat", "Crime statistic:",
                  choices = c("Murder (per 100,000)" = "Murder",
                              "Assault (per 100,000)" = "Assault",
                              "Rape (per 100,000)" = "Rape"),
                  selected = "Murder"),
      sliderInput("urbanpop", "Minimum urban population (%):",
                  min = 0, max = 100, value = 0, step = 5)
    ),

    mainPanel(
      plotOutput("barPlot", height = "600px"),
      tableOutput("stateTable")
    )
  )
)

server <- function(input, output) {

  filtered_states <- reactive({
    us_arrests %>% filter(UrbanPop >= input$urbanpop)
  })

  output$barPlot <- renderPlot({
    df <- filtered_states()

    stat_label <- switch(input$stat,
                          Murder  = "Murder arrests (per 100,000)",
                          Assault = "Assault arrests (per 100,000)",
                          Rape    = "Rape arrests (per 100,000)")

    ggplot(df, aes(x = reorder(State, .data[[input$stat]]), y = .data[[input$stat]])) +
      geom_col(fill = "#E07B39") +
      coord_flip() +
      labs(title = paste(stat_label, "by State"),
           x = "State",
           y = stat_label,
           caption = "Source: USArrests dataset (1973), R datasets package") +
      theme_minimal()
  })

  output$stateTable <- renderTable({
    filtered_states() %>%
      select(State, Murder, Assault, Rape, UrbanPop) %>%
      arrange(desc(.data[[input$stat]]))
  })

}

uspop_df <- data.frame(
  year = as.numeric(time(uspop)),
  population = as.numeric(uspop)
)

ui <- fluidPage(
  titlePanel("US Population by Decade (1790–1970)"),

  sidebarLayout(
    sidebarPanel(
      sliderInput("yearrange", "Year range:",
                  min = min(uspop_df$year), max = max(uspop_df$year),
                  value = c(min(uspop_df$year), max(uspop_df$year)),
                  step = 10, sep = ""),
      checkboxInput("showpoints", "Show data points", value = TRUE)
    ),

    mainPanel(
      plotOutput("popPlot"),
      tableOutput("popTable")
    )
  )
)

server <- function(input, output) {

  filtered_pop <- reactive({
    uspop_df %>%
      filter(year >= input$yearrange[1], year <= input$yearrange[2]) %>%
      mutate(growth_pct = round((population / lag(population) - 1) * 100, 1))
  })

  output$popPlot <- renderPlot({
    df <- filtered_pop()

    p <- ggplot(df, aes(x = year, y = population)) +
      geom_line(color = "#8e1e81", linewidth = 1.1) +
      labs(title = "US Population Over Time",
           x = "Year",
           y = "Population (millions)",
           caption = "Source: uspop dataset, R datasets package") +
      theme_minimal()

    if (input$showpoints) {
      p <- p + geom_point(color = "#8e1e81", size = 2.5)
    }

    p
  })

  output$popTable <- renderTable({
    filtered_pop() %>%
      rename(Year = year, `Population (millions)` = population,
             `Growth from prior decade (%)` = growth_pct)
  })

}

Part 4.

An app on your own data

budget_raw <- read_excel("GRB2025.xlsx", skip = 3)

names(budget_raw)[1:4] <- c("Sector", "Projects", "Budget2025", "Budget2024")

budget <- budget_raw %>%
  select(Sector, Projects, Budget2025, Budget2024) %>%
  filter(!is.na(Sector), Sector != "Total", !grepl("Exchange rate", Sector)) %>%
  mutate(
    Change = round(Budget2025 - Budget2024, 2),
    PctChange = round((Budget2025 / Budget2024 - 1) * 100, 1)
  )

ui <- fluidPage(
  titlePanel("Seoul 2025 Gender-Responsive Budgeting (GRB) by Sector"),
  
  wellPanel(
    strong("About: "),
    "Source: Seoul City 2025 Gender-Responsive Budgeting (GRB), Functional Summary Table. ",
    "Data as of 2025 (exchange rate: 1,421.5 KRW per USD). ",
    "GRB allocates a designated portion of the city budget specifically toward gender-equality outcomes, so these figures reflect that targeted GRB allocation — not Seoul's total metropolitan budget. ",
    "This app compares GRB budget allocations (2024 vs. 2025) and the number of funded GRB projects (2025) across 13 sectors."
  ),
  
  sidebarLayout(
    sidebarPanel(
      selectInput("metric", "View:",
                  choices = c("Budget comparison (2024 vs 2025)" = "compare",
                              "Number of projects (2025)" = "Projects",
                              "Budget change, 2024→2025" = "Change")),
      checkboxGroupInput("sectors", "Sectors to show:",
                         choices = unique(budget$Sector),
                         selected = unique(budget$Sector)),
      radioButtons("sortorder", "Sort order:",
                   choices = c("Descending" = "desc", "Ascending" = "asc"),
                   selected = "desc"),
      downloadButton("downloadData", "Download filtered data (CSV)")
    ),
    
    mainPanel(
      tabsetPanel(
        tabPanel("Plot", plotOutput("budgetPlot", height = "550px")),
        tabPanel("Table", tableOutput("budgetTable"))
      )
    )
  )
)

server <- function(input, output) {
  
  filtered_budget <- reactive({
    df <- budget %>% filter(Sector %in% input$sectors)
    sort_var <- if (input$metric == "compare") "Budget2025" else input$metric
    if (input$sortorder == "desc") {
      df %>% arrange(desc(.data[[sort_var]]))
    } else {
      df %>% arrange(.data[[sort_var]])
    }
  })
  
  metric_label <- reactive({
    switch(input$metric,
           compare  = "Budget (USD million)",
           Projects = "Number of projects",
           Change   = "Budget change, 2024→2025 (USD million)")
  })
  
  output$budgetPlot <- renderPlot({
    df <- filtered_budget()
    sector_order <- df$Sector
    
    if (input$metric == "compare") {
      df_long <- df %>%
        select(Sector, Budget2024, Budget2025) %>%
        pivot_longer(cols = c(Budget2024, Budget2025),
                     names_to = "Year", values_to = "Budget") %>%
        mutate(Year = recode(Year, Budget2024 = "2024", Budget2025 = "2025"),
               Sector = factor(Sector, levels = rev(sector_order)))
      
      ggplot(df_long, aes(x = Sector, y = Budget, fill = Year)) +
        geom_col(position = position_dodge(width = 0.75), width = 0.7) +
        scale_fill_manual(values = c("2024" = "#A9C6e8", "2025" = "#2E6F94")) +
        coord_flip() +
        labs(title = "GRB Budget by Sector: 2024 vs 2025",
             x = "Sector",
             y = "Budget (USD million)",
             fill = "Year",
             caption = "Source: Seoul City 2025 Gender-Responsive Budgeting (GRB) Functional Summary Table") +
        theme_minimal()
      
    } else {
      df2 <- df %>% mutate(Sector = factor(Sector, levels = rev(sector_order)))
      
      ggplot(df2, aes(x = Sector, y = .data[[input$metric]])) +
        geom_col(fill = "#2E6F94") +
        coord_flip() +
        labs(title = paste(metric_label(), "by Sector"),
             x = "Sector",
             y = metric_label(),
             caption = "Source: Seoul City 2025 Gender-Responsive Budgeting (GRB) Functional Summary Table") +
        theme_minimal()
    }
  })
  
  output$budgetTable <- renderTable({
    filtered_budget() %>%
      rename(
        Sector = Sector,
        `No. of Projects` = Projects,
        `2025 Budget (USD M)` = Budget2025,
        `2024 Budget (USD M)` = Budget2024,
        `Change (USD M)` = Change,
        `% Change` = PctChange
      )
  })
  
  output$downloadData <- downloadHandler(
    filename = function() "GRB2025_filtered.csv",
    content = function(file) {
      write.csv(filtered_budget(), file, row.names = FALSE)
    }
  )
}
